# 세종 탁구플랫폼 Cloudflare Pages 배포

cd $env:USERPROFILE\Documents\sejong-takgu-pages
git pull origin main
Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass

# 대용량 mp4 제외한 dist 빌드 후 배포 (25MB 제한 회피)
npx.cmd wrangler pages deploy dist --project-name=sejong-takgu --commit-dirty=true
